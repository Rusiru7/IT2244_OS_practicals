#include<stdio.h>
int main(){
        int age=25;
        printf("%d\n",age);

        age=31;
        printf("New age:%d\n",age);

        int a=5;
        int b=a;
        printf("A: %d\n",a);
        printf("B: %d",b);
return 0;
}
