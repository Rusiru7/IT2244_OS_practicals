//Q2 - Factorial
output : Factorial of 5 is : 120


//code
echo "Enter the value: "
read num
fac=1
for((i=1; i<=num; i++))
do
fac=$(($fac*$i))
done
echo 'Factorial of ' $num ' is '$fac

//output
asiri@DESKTOP-0MK28KD:~$ vi fact.sh
asiri@DESKTOP-0MK28KD:~$ chmod 777 fact.sh
asiri@DESKTOP-0MK28KD:~$ ./fact.sh
